public class Calculator {

    public static int calculate(int a, int b, char op) {
        switch(op) {
            case '+': return a + b;
            case '-': return a - b;
            case '*': return a * b;
            case '/':
                if (b != 0) return a / b;
                else throw new ArithmeticException("Divide by zero");
            default:
                throw new IllegalArgumentException("Invalid operator");
        }
    }

    // Adding the calling function here
    public static void main(String[] args) {
        int num1 = 10;
        int num2 = 5;
        char operator = '*';

        // Call the function and print the output
        int result = calculate(num1, num2, operator);
        
        System.out.println(num1 + " " + operator + " " + num2 + " = " + result);
    }
}

public class BankAccount {

    double balance = 0;

    public void deposit(double amount) {
        balance += amount;
    }

    public boolean withdraw(double amount) {
        if (amount <= balance) {
            balance -= amount;
            return true;
        }
        return false;
    }

    public double getBalance() {
        return balance;
    }

    public static void main(String[] args) {
        // 1. Create an instance of the account
        BankAccount myAccount = new BankAccount();

        // 2. Call deposit
        myAccount.deposit(500.0);
        System.out.println("Balance after deposit: " + myAccount.getBalance());

        // 3. Call withdraw and check if it worked
        boolean success = myAccount.withdraw(200.0);
        if (success) {
            System.out.println("Withdrawal successful!");
        } else {
            System.out.println("Insufficient funds!");
        }

        // 4. Show final balance
        System.out.println("Final Balance: " + myAccount.getBalance());
    }
}

import static org.junit.Assert.*;
import org.junit.Test;

public class BankAccountTest {

    @Test
    public void testDeposit() {
        BankAccount acc = new BankAccount();
        acc.deposit(100);
        assertEquals(100, acc.getBalance(), 0.001);
    }

    @Test
    public void testWithdrawSuccess() {
        BankAccount acc = new BankAccount();
        acc.deposit(100);
        assertTrue(acc.withdraw(50));
        assertEquals(50, acc.getBalance(), 0.001);
    }

    @Test
    public void testWithdrawFail() {
        BankAccount acc = new BankAccount();
        acc.deposit(50);
        assertFalse(acc.withdraw(100));
    }
}
